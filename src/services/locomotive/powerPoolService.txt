import type { Locomotive, Train } from "@/types";

export interface PowerPoolSelection {
  locomotives: Locomotive[];
  totalHorsepower: number;
  totalAxles: number;
}

export interface PowerPoolRequest {
  train: Train;
  availableLocomotives: Locomotive[];
  minimumHorsepower: number;
  minimumAxles?: number;
  preferredModels?: string[];
  allowForeignPower?: boolean;
}

export function getAvailableLocomotivesAtYard(
  locomotives: Locomotive[],
  yard: string,
): Locomotive[] {
  return locomotives.filter(
    (l) =>
      l.status === "Available" &&
      l.currentYard === yard &&
      !l.currentTrainId,
  );
}

export function choosePower(
  request: PowerPoolRequest,
): PowerPoolSelection | null {
  let candidates = getAvailableLocomotivesAtYard(
    request.availableLocomotives,
    request.train.origin,
  );

  if (!request.allowForeignPower) {
    candidates = candidates.filter(
      (l) => l.roadName === "ISR",
    );
  }

  if (request.preferredModels?.length) {
    const preferred = candidates.filter((l) =>
      request.preferredModels!.includes(l.model),
    );

    if (preferred.length) {
      candidates = preferred;
    }
  }

  candidates = [...candidates].sort(
    () => Math.random() - 0.5,
  );

  const selected: Locomotive[] = [];

  let horsepower = 0;
  let axles = 0;

  for (const loco of candidates) {
    selected.push(loco);

    horsepower += loco.horsepower;
    axles += loco.axles;

    if (
      horsepower >= request.minimumHorsepower &&
      (!request.minimumAxles ||
        axles >= request.minimumAxles)
    ) {
      break;
    }
  }

  if (horsepower < request.minimumHorsepower)
    return null;

  return {
    locomotives: selected,
    totalHorsepower: horsepower,
    totalAxles: axles,
  };
}
